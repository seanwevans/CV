\section*{Selected Research \& Engineering Projects}

\begin{itemize}

    \item
    \project{\href{https://github.com/seanwevans/pynq_butterfly}
    {pynq\_butterfly}}
    {Exact FPGA accelerator for OpenFHE BGVRNS ciphertext multiplication and BV
    relinearization on a PYNQ-Z2. Achieved 245.61 exact relinearized
    ciphertexts/sec at 98.84\% of the calculated transport ceiling; validated
    6,291,456 residue outputs against OpenFHE with zero mismatches.}

    \item
    \project{\href{https://github.com/seanwevans/lockstep}{Lockstep}}
    {Data-oriented systems programming language and compiler for deterministic,
    high-throughput compute pipelines. Implements straight-line SIMD execution,
    static memory topology, semantic type checking, LLVM IR generation, manual
    vector lowering, benchmarking, and LSP support.}

    \item
    \project{\href{https://github.com/seanwevans/fluid-sims}{fluid-sims}}
    {CUDA-accelerated numerical simulation laboratory spanning SPH fluids,
    hypersonic flow, reaction-diffusion, Burgers flow, shallow-water systems,
    and 3-D fluid dynamics.}

    \item
    \project{\href{https://github.com/seanwevans/pg_gpt2}{pg\_gpt2}}
    {Complete GPT-2 implementation inside PostgreSQL, including a native C tensor
    engine, relational reverse-mode autodiff, AdamW optimization, checkpointing,
    tokenization, training, and inference through SQL.}

    \item
    \project{\href{https://github.com/seanwevans/WarpDB}{WarpDB}}
    {GPU analytical query engine that parses SQL-like expressions into ASTs and
    JIT-compiles them into CUDA kernels using NVRTC, with columnar-data and
    multi-GPU execution support.}

\end{itemize}
